%% AULA 01.EX01 - Exemplos basicos de PROLOG
%% fonte: http://www.bitwisemag.com/copy/programming/prolog/intro/firststeps.html

?- write('Hello world'), nl.  %% nl = newline

?- 4 is 2 + 2.    %% is primeiro valora 2 + 2 e depois compara se eh igual a 4
true.

?- 4 is 2 + 3.
false.
 
?- 4 = 4.          %% = verifica se os simbolos sao iguais
true.

?- 4 = 2 + 2.      %% falso, pois os simbolos sao diferentes
false.

%% predicados e simbolos iniciam com letra minuscula
%% variaveis com maiusculas

?- X is 2, write(X).  %% teste com variavel
2
X = 2.

?- X is *(3, 4), write(X). %% operacao aritmetica prefix
